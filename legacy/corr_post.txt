const STAT_LABEL={individual:'Individual / event value',minimum:'Minimum',mean:'Mean / average',median:'Median',maximum:'Maximum',category:'Category'};
function esc(x){return String(x??'').replace(/[&<>"']/g,c=>({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'}[c]))}
function finite(x){return typeof x==='number'&&Number.isFinite(x)}
function metric(id){return DATA.metrics.find(x=>x.metric_id===id)}
function eventRecords(eid,mid,stat){return DATA.records.filter(r=>r.target_id===eid&&r.metric_id===mid&&r.statistic_type===stat)}
function metricRecords(mid){return DATA.records.filter(r=>r.metric_id===mid)}
function median(v){if(!v.length)return null;v=[...v].sort((a,b)=>a-b);return v.length%2?v[(v.length-1)/2]:(v[v.length/2-1]+v[v.length/2])/2}
function eventValue(eid,mid,stat){const a=eventRecords(eid,mid,stat),n=a.map(r=>r.value).filter(finite);return n.length?median(n):(a.map(r=>typeof r.value==='string'?r.value:null).find(Boolean)??null)}
function availableStats(mid){return [...new Set(metricRecords(mid).map(r=>r.statistic_type))].sort((a,b)=>['individual','minimum','mean','median','maximum','category'].indexOf(a)-['individual','minimum','mean','median','maximum','category'].indexOf(b))}
function fillStats(sel,mid,preferred){const a=availableStats(mid);sel.innerHTML='';a.forEach(s=>{const o=document.createElement('option');o.value=s;o.textContent=STAT_LABEL[s]||s;sel.appendChild(o)});sel.value=a.includes(preferred)?preferred:a[0]}
function fmt(v,u=''){return finite(v)?Number(v.toPrecision(6))+(u?' '+u:''):esc(v)}
